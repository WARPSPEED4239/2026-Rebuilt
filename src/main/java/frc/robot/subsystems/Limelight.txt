package frc.robot.subsystems;

import edu.wpi.first.wpilibj.smartdashboard.SmartDashboard;
import edu.wpi.first.wpilibj2.command.SubsystemBase;
import frc.robot.Constants;
import frc.robot.LimelightHelpers;

public class Limelight extends SubsystemBase {
  
  private final LimelightHelpers ClimberLimelight = new LimelightHelpers();
  private final double KpSIDE = 0.075; //Tune; too high will cause oscillations

  public Limelight() {}

  public double GetLeftRightSpeed(double maxDriveSpeed, double tx) {

    double kP = 0.05;

    double speed = -tx * kP;

    speed *= maxDriveSpeed;

    // minimum speed so robot actually moves
    if (Math.abs(speed) < 0.1 && Math.abs(tx) > 0.5) {
        speed = Math.copySign(0.1, speed);
    }

    // clamp
    speed = Math.max(-maxDriveSpeed, Math.min(maxDriveSpeed, speed));

    return speed;
}

  public double getTX() {
    return ClimberLimelight.getTX(Constants.LIMELIGHT_NAME);
  }

  public double getTA() {
    if(!ClimberLimelight.getTV(Constants.LIMELIGHT_NAME)) return;
    return ClimberLimelight.getTA(Constants.LIMELIGHT_NAME);
  }

  public double TXOffsetRight(double TA){
    return (TA -1.56) / -0.0615;
  }

  @Override
  public void periodic() {
    SmartDashboard.putNumber("Limelight Distance", getTA());
  }
  
}
